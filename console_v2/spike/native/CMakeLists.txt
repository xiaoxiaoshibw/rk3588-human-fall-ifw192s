cmake_minimum_required(VERSION 3.21)
project(ConsoleV2Spike LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Quick Quick3D)
qt_add_executable(QtSpike WIN32 main.cpp PointGeometry.cpp PointGeometry.h ProbeApplication.h)
target_compile_options(QtSpike PRIVATE -Wall -Wextra -Werror)
target_link_libraries(QtSpike PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Quick3D)
qt_add_resources(QtSpike "qml" PREFIX "/" FILES Main.qml)
