cmake_minimum_required(VERSION 3.21)
project(ConsoleV2App LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Quick Quick3D QuickControls2 Network Test)
qt_add_executable(ConsoleV2 WIN32 main.cpp Viewer.h AnnotationView.h Preview.h)
target_compile_options(ConsoleV2 PRIVATE -Wall -Wextra -Werror)
target_link_libraries(ConsoleV2 PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Quick3D Qt6::QuickControls2 Qt6::Network Qt6::Test)
qt_add_resources(ConsoleV2 "qml" PREFIX "/" FILES Main.qml colormap_depth.png colormap_height.png colormap_intensity.png)
