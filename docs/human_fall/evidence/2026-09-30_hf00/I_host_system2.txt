#!/bin/bash
# lidar-stack.sh - 3588 开机自启：动态定位雷达网口 + 容器 + ROS 驱动 + foxglove_bridge
# 雷达物理网口按 MAC 识别，避免将联网的 eth1 误当雷达口；接口名可以变化
set -e

# [1/4] 寻找指定 MAC 且有载波的雷达网口
LIDAR_IF=""
LIDAR_MAC="08:e2:6d:c4:fd:47"  # radar physical port; name may change on reboot
for attempt in 1 2 3 4 5 6; do
  for iface in $(ls /sys/class/net/ | grep -E "^eth"); do
    [ "$(cat /sys/class/net/$iface/address 2>/dev/null)" = "$LIDAR_MAC" ] || continue
    if [ "$(cat /sys/class/net/$iface/operstate 2>/dev/null)" = "up" ]; then
      LIDAR_IF="$iface"
      break
    fi
  done
  [ -n "$LIDAR_IF" ] && break
  sleep 5
done

if [ -z "$LIDAR_IF" ]; then
  echo "[1/4] WARNING: 未检测到指定 MAC 的雷达网口或未接线，跳过雷达 IP 配置"
else
  echo "[1/4] 雷达网口: $LIDAR_IF"
  ip link set "$LIDAR_IF" up 2>/dev/null || true
  # 幂等配置：已存在则忽略，不存在则添加
  if ! ip addr show "$LIDAR_IF" | grep -q "192.168.1.35"; then
    ip addr add 192.168.1.35/24 dev "$LIDAR_IF" 2>/dev/null || true
  fi
  if ! ip route show | grep -q "192.168.1.10 dev $LIDAR_IF"; then
    ip route replace 192.168.1.10/32 dev "$LIDAR_IF" src 192.168.1.35
  fi
fi

if ! docker ps --format '{{.Names}}' | grep -qx slam-localization; then
    docker start slam-localization
fi
sleep 5
echo "[2/4] 容器已启动"

docker exec slam-localization bash -c '
  if ! pgrep -f "^/usr/bin/python3 /opt/ros/noetic/bin/rosmaster" > /dev/null; then
    source /opt/ros/noetic/setup.bash
    roscore > /tmp/roscore.log 2>&1 &
    sleep 5
  fi'
echo "[3/4] roscore 就绪"

docker exec slam-localization bash -c '
  source /opt/ros/noetic/setup.bash
  source /root/catkin_ws/devel/setup.bash
  # 清掉残留/僵尸驱动进程（锚定二进制路径，避免误杀本脚本 shell）
  for p in $(pgrep -f "^/root/catkin_ws/devel/lib/inno_lidar_ros/inno_lidar_node"); do kill -9 "$p" 2>/dev/null || true; done
  if ! pgrep -f "^/root/catkin_ws/devel/lib/inno_lidar_ros/inno_lidar_node" > /dev/null; then
    rosrun inno_lidar_ros inno_lidar_node _config_path:=/root/catkin_ws/src/inno_lidar_ros/config/config.yaml > /tmp/lidar.log 2>&1 &
  fi
  if ! pgrep -f "^/opt/ros/noetic/lib/foxglove_bridge/foxglove_bridge" > /dev/null; then
    rosrun foxglove_bridge foxglove_bridge _port:=8765 _send_buffer_limit_bytes:=134217728 _num_sender_threads:=4 > /tmp/bridge.log 2>&1 &
  fi'
echo "[4/4] 驱动 + bridge 已启动"
